# ScrollArea.vue 竖向

<template>
	<div>
		<div class="wrapper " ref="wrapper1"  style="overflow: hidden;" v-loading="loading">
			<div   ref="content">
				<div >
					<slot></slot>
				</div>
			</div>
		</div>
	</div>
</template>
<script>

	import BScroll from 'better-scroll'

	export default{
		components:{},
		props:[],
		data(){
			return {
				scroll:{},
				loading:true
			}
		},
		methods:{
scrollToElement:function(obj){
    this.scroll.scrollToElement(obj,500);
   },
   refresh:function(){ 
    if(this.scroll.refresh){
     this.scroll.refresh(); 
    }else{
     this.initScrollArea();
    }
   },

			initScrollArea:function(){
				if(this.scroll.refresh){
					this.scroll.refresh();
					return false;
				}
				// var area = this.$refs.content;

				// if(area.offsetHeight <= this.$refs.wrapper1.offsetHeight){
				// 	this.loading = false;

				// 	return false;
				// }

				this.$nextTick(()=>{

					this.scroll = new BScroll(this.$refs.wrapper1, {
							 //  	pullUpLoad: {
								//   threshold: -20 // 在上拉到超过底部 20px 时，触发 pullingUp 事件
								// },
							  	//开启点击事件 默认为false
							  	click:true,
							  	scrollbar: {
							  	 fade: false,
							  	 interactive: false // 1.8.0 新增
							  	},
							  	mouseWheel: {
							  	 speed: 20,
							  	 invert: false,
							  	 easeTime: 300
							  	},
							  	probeType:3

							    })

					this.loading = false;

				})
			}
		},
		mounted(){
			//this.$nextTick(()=>{

			//	setTimeout(this.initScrollArea,300);

			//});

// 当事情发生变化时调用`update`
      this.observer = new MutationObserver(this.refresh);

      // 观察这个组件的变化
      this.observer.observe(this.$el, {
        childList: true,
        subtree: true
      });

		},
beforeUnmount() {
      this.observer.disconnect();
    },

		created(){

		},
		computed:{

		},
		watch:{

		}
	}

</script>
<style scoped>

.wrapper{
	  position:absolute; 
	  left:0;
	  top:0;
	  right:0;
	  bottom: 0; 
	} 

</style>
<style>
	.wrapper .bscroll-vertical-scrollbar{
		z-index: 1 !important;
	}
</style>
