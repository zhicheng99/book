# 分块上传视频 对接ali-oss

调用

<UploadByChunkAliOss
:upload-video.sync="addForm.videoUrl"
:piece-size="1024000" >< /UploadByChunkAliOss>

 <template>
	<div>
		<el-input size="small" v-model="uploadVideo" @input="input" placeholder="上传视频或粘贴视频地址"> 

			<template slot="append">
				<el-upload
			action="/admin/file/uploadVideoFile?module=dc_video"
			:auto-upload="false"
			limit="1"
			:show-file-list="false"
			:on-success="handletopSuccess"
			:on-change="fileChange" 
			:before-upload="handleBeforeUpload"
			:on-error="handleUploadError"
			ref="upload"
			accept=".mov,.mp4,.m4a,.3gp,.3g2"
		>
			<el-button size="small" type="primary" :loading="dialogLoaidng"> 
			{{percent!=0?percent+'%':''}}选择视频</el-button>
		</el-upload>
			</template>

		</el-input>
	</div>
</template>
<script>

let OSS = require('ali-oss')

let ossConfig = {
  // region以杭州为例（oss-cn-hangzhou），其他region按实际情况填写。
  region: '',
  // 阿里云主账号AccessKey拥有所有API的访问权限，风险很高。强烈建议您创建并使用RAM账号进行API访问或日常运维，请登录RAM控制台创建RAM账号。
  accessKeyId: '',
  accessKeySecret: '',
  bucket: ''
}

let client = new OSS(ossConfig);
console.log(client);

let tempCheckpoint;


	export default{
		components:{},
		props:['uploadVideo','uploadCover','pieceSize'],
		data(){
			return {
				dialogLoaidng:false,

				showVideo: false,
	            blobSlice: null,
	            file: null,
	            identifier: null,
	            chunkSize: 1024000,
	            chunks: 0,
	            currentChunk: 0,
	            spark: null,
	            fileReader: null,
	            tmpDataList: [],
	            formDataList: [],
	            uploadedList: [],
	            start: 0,
	            end: 0,
	            headers: {},
	            // urlCode: code.urlCode.lectureDemand,
	            xhr: null,
	            pauseStatus: false,
	            showProgress: false,
	            percent: 0,


	            uploadingChunk:0,  //正在上传的块

			}
		},
		methods:{
			input:function(){
				this.$emit('update:uploadVideo',this.uploadVideo);

			},
			getExt:function(name){
				var tmp = name.split('.');

				return tmp[tmp.length-1];
			}, 
			fileChange:function(file){
				this.file = file.raw;
				var t = new Date()
				var fileName = t.getFullYear()+''+(t.getMonth()+1)+''+t.getDate()+''+parseInt(Math.random()*10000)+''+this.file.uid;
				var folderName = t.getFullYear()+''+(t.getMonth()+1)+''+t.getDate()+'/video/';

if(file.size <= this.pieceSize){  //小于分片的
	
					client.put(folderName+fileName+'.'+this.getExt(this.file.name), this.file).then(res=>{
						this.$emit('update:uploadVideo',res.url);
					})

				}else{
 

				client.multipartUpload(folderName+fileName+'.'+this.getExt(this.file.name), this.file, { 
			      progress:  (p, checkpoint,res)=> {
			      	this.percent = parseInt(p*100);


			      	if(this.percent == 100){  //上传完成
			      		var tmp = res.requestUrls[0].split('?');
						this.$emit('update:uploadVideo',tmp[0]);

						//视频截封面图
						this.$post('/admin/file/getVideoscreenshots',{videoUrl:tmp[0]});

						this.$refs.upload.clearFiles();

						setTimeout(()=>{

							this.percent = 0;

						},2000);

			      	}

			        // 断点记录点。浏览器重启后无法直接继续上传，您需要手动触发上传操作。
			        // tempCheckpoint = checkpoint;
			      },
			      // meta: { year: 2020, people: 'test' },
			      mime: this.file.type,
			      parallel: 1,
				  partSize: this.chunkSize
				  // callback:(url)=>{
				  // 	console.log(url);
				  // }
			   })
}

 
			}, 

		},
		mounted(){

		},
		created(){
			this.blobSlice =
	            window.File.prototype.slice ||
	            window.File.prototype.mozSlice ||
	            window.File.prototype.webkitSlice;
	        // this.spark = new SparkMD5.ArrayBuffer();
	        this.fileReader = new FileReader();
	        this.percent = 0;
		},
		computed:{

		},
		watch:{
			'pieceSize':{
				handler:function(v){
					console.log(typeof v);
					if(typeof v !='undefined'){

						if(v <=1024000){
							this.chunkSize = 1024000
						}else{
							this.chunkSize = v;
						}
					}

				},
				immediate:true
			}
		}
	}

</script>
