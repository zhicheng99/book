# 部分前端面试题参考答案

# 二、

const url = 'http://example.com/index.html?a=1&b=2&c=3&d';
参考结果
const result = {
a: '1',
b: '2',
c: '3',
d: true,
}

参考答案：

function solve(s){

    var s = s.replace(/.*?\?/g,'');

    var arr = s.split('&'); 
    var tmp = {};

    for(var i=0;i<arr.length;i++){

        if(arr[i].indexOf('=')>-1){

            var t = arr[i].split('=');


            tmp[t[0]] = t[1]!=''?t[1]:true;


        }else{

            tmp[arr[i]] = true;

        }

    }

    return tmp;

}



document.write(JSON.stringify(solve(url)));

# 五、

const data = [
    {
        "id": 1,
        "geom_xy": "POINT (0 0)", 
        "geom_z": "0"
    },
    {
        "id": 2,
        "geom_xy": "LINESTRING (1 1, 2 2, 3 3)", 
        "geom_z": "1,2,3"
    },
    {
        "id": 3,
        "geom_xy": "POLYGON ((4 4, 5 5, 6 6, 7 7, 4 4))", 
        "geom_z": "4,5,6,7,4"
    }

]
把data转换为以下格式
const result = [
{"id":1,
"geom":{"type":"POINT","coordinates":[[0,0,0]]}
},
{"id":2,
"geom":{"type":"LINESTRING","coordinates":[[1,1,1],[2,2,2],[3,3,3]]}
},
{"id":3,
"geom":{"type":"POLYGON","coordinates":[[4,4,4],[5,5,5],[6,6,6],[7,7,7],[4,4,4]]}
}
]

参考答案：

function solve(data){

    var getType = function(data){
        return data.split(' ')[0];
    }

    var getCoordinates = function(obj){
        var xy_tmp = obj.geom_xy.replace(/.*?\(/g,'').replace(/(\)|\()/g,'');
        var arr_xy = xy_tmp.split(', ');
        var arr_z = obj.geom_z.split(',');
  
        var tmp = [];
        for(var i = 0;i<arr_z.length; i++){
   
                tmp[i]=[arr_z[i]*1]; 
  
            var t = arr_xy[i].split(' ');
            for(var j=0; j<t.length;j++){
                tmp[i].push(t[j]*1);
            }
        }

        return tmp;
    }

    var tmp = [];
    for(var i in data){
        tmp.push({
            id:data[i].id,
            geom:{
                type:getType(data[i].geom_xy),
                coordinates:getCoordinates(data[i])
            }
        })

    }
    return tmp;
} 
document.write(JSON.stringify(solve(data)));
